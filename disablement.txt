BEGIN;

UPDATE public."AO_8542F1_IFJ_ATM_RULE"
SET "DISABLED" = true
WHERE "DISABLED" IS DISTINCT FROM true;

UPDATE public."AO_8542F1_IFJ_IMPORT_SRC"
SET "SYNCHRONIZATION_ENABLED" = false
WHERE "SYNCHRONIZATION_ENABLED" IS DISTINCT FROM false;

SELECT
    "DISABLED",
    count(*)
FROM public."AO_8542F1_IFJ_ATM_RULE"
GROUP BY "DISABLED"
ORDER BY "DISABLED";

SELECT count(*) AS enabled_assets_automation
FROM public."AO_8542F1_IFJ_ATM_RULE"
WHERE "DISABLED" IS DISTINCT FROM true;

SELECT count(*) AS scheduled_assets_imports
FROM public."AO_8542F1_IFJ_IMPORT_SRC"
WHERE "SYNCHRONIZATION_ENABLED" IS DISTINCT FROM false;

SELECT
    "STATUS",
    "SYNCHRONIZATION_ENABLED",
    count(*)
FROM public."AO_8542F1_IFJ_IMPORT_SRC"
GROUP BY "STATUS", "SYNCHRONIZATION_ENABLED"
ORDER BY 1,2;

COMMIT;
