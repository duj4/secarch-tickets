oc exec -n <namespace> <pod> -- sh -c '
echo "===== cgroup current ====="
cat /sys/fs/cgroup/memory.current 2>/dev/null || true

echo "===== cgroup breakdown ====="
grep -E "^(anon|file|kernel|slab|slab_reclaimable|slab_unreclaimable|sock|shmem|file_mapped|file_dirty) " \
  /sys/fs/cgroup/memory.stat 2>/dev/null || true

echo "===== process memory ====="
grep -E "^(Rss|Pss|Anonymous|Private_Dirty|Shared_Clean|Swap):" \
  /proc/1/smaps_rollup 2>/dev/null || true

echo "===== storage ====="
du -sh /var/log/fluent-bit-storage 2>/dev/null || true
find /var/log/fluent-bit-storage -type f 2>/dev/null | wc -l
'
