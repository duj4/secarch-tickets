BEGIN READ ONLY;

SET LOCAL statement_timeout = '120s';
SET LOCAL lock_timeout = '2s';

WITH docs AS (
    SELECT
        'current'::text AS source,
        d.id AS dashboard_id,
        NULL::bigint AS dashboard_version_id,
        NULL::int AS version,
        d.org_id,
        d.uid AS dashboard_uid,
        d.title,
        d.data::jsonb AS data
    FROM dashboard d

    UNION ALL

    SELECT
        'history'::text AS source,
        dv.dashboard_id,
        dv.id AS dashboard_version_id,
        dv.version,
        d.org_id,
        d.uid AS dashboard_uid,
        d.title,
        dv.data::jsonb AS data
    FROM dashboard_version dv
    JOIN dashboard d
      ON d.id = dv.dashboard_id
),
target_containers AS (
    SELECT
        d.*,
        x.targets
    FROM docs d
    CROSS JOIN LATERAL jsonb_path_query(
        d.data,
        'strict $.**.targets'
    ) AS x(targets)
),
target_items AS (
    -- Normal targets array
    SELECT
        tc.source,
        tc.dashboard_id,
        tc.dashboard_version_id,
        tc.version,
        tc.org_id,
        tc.dashboard_uid,
        tc.title,
        a.item
    FROM target_containers tc
    CROSS JOIN LATERAL jsonb_array_elements(tc.targets)
        AS a(item)
    WHERE jsonb_typeof(tc.targets) = 'array'

    UNION ALL

    -- Grafana also accepts targets as an object
    SELECT
        tc.source,
        tc.dashboard_id,
        tc.dashboard_version_id,
        tc.version,
        tc.org_id,
        tc.dashboard_uid,
        tc.title,
        o.value
    FROM target_containers tc
    CROSS JOIN LATERAL jsonb_each(tc.targets)
        AS o(key, value)
    WHERE jsonb_typeof(tc.targets) = 'object'
)

SELECT
    source,
    org_id,
    dashboard_id,
    dashboard_version_id,
    version,
    dashboard_uid,
    title,
    item -> '' AS empty_key_value
FROM target_items
WHERE jsonb_typeof(item) = 'object'
  AND item ? ''
ORDER BY
    org_id,
    dashboard_uid,
    source,
    version;

ROLLBACK;
