BEGIN READ ONLY;

SET LOCAL statement_timeout = '120s';
SET LOCAL lock_timeout = '2s';

SELECT
    d.org_id,
    d.id AS dashboard_id,
    d.uid AS dashboard_uid,
    d.title
FROM dashboard d
WHERE EXISTS (
    SELECT 1
    FROM jsonb_path_query(
        d.data::jsonb,
        'strict $.**.targets'
    ) AS t(targets)
    CROSS JOIN LATERAL (
        SELECT x.item
        FROM jsonb_array_elements(
            CASE
                WHEN jsonb_typeof(t.targets) = 'array'
                THEN t.targets
                ELSE '[]'::jsonb
            END
        ) AS x(item)

        UNION ALL

        SELECT x.value
        FROM jsonb_each(
            CASE
                WHEN jsonb_typeof(t.targets) = 'object'
                THEN t.targets
                ELSE '{}'::jsonb
            END
        ) AS x(key, value)
    ) AS target
    WHERE jsonb_typeof(target.item) = 'object'
      AND target.item ? ''
)
ORDER BY d.org_id, d.uid;

ROLLBACK;
