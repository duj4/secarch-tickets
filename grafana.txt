
BEGIN READ ONLY;

SET LOCAL statement_timeout = '120s';
SET LOCAL lock_timeout = '2s';

WITH docs AS (
    SELECT
        'current'::text AS source,
        d.id AS dashboard_id,
        NULL::bigint AS dashboard_version_id,
        NULL::int AS version,
        d.org_id,
        d.uid AS dashboard_uid,
        d.title,
        d.data::jsonb AS data
    FROM dashboard d

    UNION ALL

    SELECT
        'history'::text,
        dv.dashboard_id,
        dv.id,
        dv.version,
        d.org_id,
        d.uid,
        d.title,
        dv.data::jsonb
    FROM dashboard_version dv
    JOIN dashboard d
      ON d.id = dv.dashboard_id
),
target_containers AS (
    SELECT
        d.*,
        x.targets
    FROM docs d
    CROSS JOIN LATERAL jsonb_path_query(
        d.data,
        'strict $.**.targets'
    ) AS x(targets)
),
target_items AS (
    SELECT
        tc.source,
        tc.dashboard_id,
        tc.dashboard_version_id,
        tc.version,
        tc.org_id,
        tc.dashboard_uid,
        tc.title,
        a.item
    FROM target_containers tc
    CROSS JOIN LATERAL jsonb_array_elements(tc.targets) AS a(item)
    WHERE jsonb_typeof(tc.targets) = 'array'

    UNION ALL

    SELECT
        tc.source,
        tc.dashboard_id,
        tc.dashboard_version_id,
        tc.version,
        tc.org_id,
        tc.dashboard_uid,
        tc.title,
        o.value
    FROM target_containers tc
    CROSS JOIN LATERAL jsonb_each(tc.targets) AS o(key, value)
    WHERE jsonb_typeof(tc.targets) = 'object'
),
bad AS (
    SELECT *
    FROM target_items
    WHERE jsonb_typeof(item) = 'object'
      AND item ? ''
)
SELECT
    org_id,
    dashboard_id,
    dashboard_uid,
    title,

    count(*) FILTER (
        WHERE source = 'current'
    ) AS current_bad_targets,

    count(DISTINCT dashboard_version_id) FILTER (
        WHERE source = 'history'
    ) AS affected_history_versions,

    count(*) FILTER (
        WHERE source = 'history'
    ) AS history_bad_targets,

    min(version) FILTER (
        WHERE source = 'history'
    ) AS first_bad_version,

    max(version) FILTER (
        WHERE source = 'history'
    ) AS last_bad_version

FROM bad
GROUP BY
    org_id,
    dashboard_id,
    dashboard_uid,
    title
ORDER BY org_id, dashboard_uid;

ROLLBACK;
