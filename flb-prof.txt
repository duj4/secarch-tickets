for pod in fluent-bit-drpvt fluent-bit-fjdtk fluent-bit-xldjt; do
  echo "========== $pod =========="

  kubectl logs "$pod" -c fluent-bit --timestamps --since=20m |
    grep 'Allocated:' | tail -2

  kubectl exec "$pod" -c fluent-bit -- sh -c '
    date -u

    echo "=== PROCESS ==="
    grep -E "^(VmRSS|RssAnon|Threads):" /proc/1/status

    echo "=== SMAPS ==="
    grep -E "^(Anonymous|AnonHugePages|Private_Dirty):" \
      /proc/1/smaps_rollup

    echo "=== CGROUP ==="
    grep -E "^(anon|anon_thp|slab_reclaimable) " \
      /sys/fs/cgroup/memory.stat
  '
done
