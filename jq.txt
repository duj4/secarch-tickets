
(
  [.items[]
    | select(
        .metadata.labels["advisor.grafana.app/type"]
        == "datasource"
      )
  ]
  | sort_by(.metadata.creationTimestamp)
  | last
) as $report

| if $report == null
     or ($report.status.report.failures | type) != "array"
  then
    error("No valid Advisor datasource report")
  else
    [{
      item: "__advisor_check__",
      itemID: "",
      url: "",
      moreInfo: "",
      failed: 0
    }]
    +
    [
      $report.status.report.failures[]
      | select(.stepID == "health-check")
      | {
          item: (.item // "unknown"),
          itemID: (.itemID // "unknown"),
          url: (.links[0].url // ""),
          moreInfo: (.moreInfo // ""),
          failed: 1
        }
    ]
  end
