{
    printf "ALTER TABLE public.\"%s\" VALIDATE CONSTRAINT \"%s\";\n",
           $1, $2
}
' /tmp/missing_ao_fk_meta.txt \
> /tmp/validate_missing_ao_fk.sql

SELECT
    c.relname AS table_name,
    con.conname,
    pg_get_constraintdef(con.oid, true)
FROM pg_constraint con
JOIN pg_class c
  ON c.oid = con.conrelid
JOIN pg_namespace n
  ON n.oid = c.relnamespace
WHERE con.contype = 'f'
  AND NOT con.convalidated
  AND n.nspname = 'public'
  AND c.relname LIKE 'AO\_%' ESCAPE '\'
ORDER BY c.relname, con.conname;
