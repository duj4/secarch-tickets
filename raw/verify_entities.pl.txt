cat > "${WORK}/verify_entities.pl" <<'PERL'
use strict;
use warnings;

my $prod_host   = $ENV{PROD_HOST}   // die "PROD_HOST is not set\n";
my $prod_attach = $ENV{PROD_ATTACH} // die "PROD_ATTACH is not set\n";
my $qa_attach   = $ENV{QA_ATTACH}   // die "QA_ATTACH is not set\n";

my $cluster      = 0;
my $mailserver   = 0;
my $mailfetcher  = 0;
my $gadget_prod  = 0;
my $prod_attach_left = 0;
my $fileattachment = 0;

while (<>) {
    $cluster++
        if /<ClusterNode(?:\s|>)/;

    $mailserver++
        if /<MailServer(?:\s|>)/;

    $mailfetcher++
        if /<ServiceConfig(?:\s|>)/ && /MailFetcherService/;

    $fileattachment++
        if /<FileAttachment(?:\s|>)/;

    $gadget_prod++
        if /<GadgetUserPreference(?:\s|>)/
        && index($_, $prod_host) >= 0;

    $prod_attach_left++
        if $prod_attach ne $qa_attach
        && index($_, $prod_attach) >= 0;
}

print "CLUSTERNODE_OUT=$cluster\n";
print "MAILSERVER_OUT=$mailserver\n";
print "MAILFETCHER_OUT=$mailfetcher\n";
print "PROD_HOST_IN_GADGET_OUT=$gadget_prod\n";
print "PROD_ATTACH_PATH_OUT=$prod_attach_left\n";
print "FILEATTACHMENT_OUT=$fileattachment\n";

exit(
    $cluster
    || $mailserver
    || $mailfetcher
    || $gadget_prod
    || $prod_attach_left
    ? 2 : 0
);
PERL
