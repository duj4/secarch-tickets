cat > "${WORK}/transform_entities.pl" <<'PERL'
use strict;
use warnings;

my $prod_host   = $ENV{PROD_HOST}   // die "PROD_HOST is not set\n";
my $qa_host     = $ENV{QA_HOST}     // die "QA_HOST is not set\n";
my $prod_attach = $ENV{PROD_ATTACH} // die "PROD_ATTACH is not set\n";
my $qa_attach   = $ENV{QA_ATTACH}   // die "QA_ATTACH is not set\n";

my $cluster_deleted       = 0;
my $mailserver_deleted    = 0;
my $mailfetcher_deleted   = 0;
my $fileattachment_count  = 0;
my $gadget_url_replaced   = 0;
my $attach_path_replaced  = 0;

while (my $line = <STDIN>) {

    $fileattachment_count++
        if $line =~ /<FileAttachment(?:\s|>)/;

    # Remove production Data Center node registrations.
    if ($line =~ /<ClusterNode(?:\s|>)/) {
        $cluster_deleted++;
        next;
    }

    # Remove Jira core incoming/outgoing mail servers.
    if ($line =~ /<MailServer(?:\s|>)/) {
        $mailserver_deleted++;
        next;
    }

    # Remove Jira core mail fetch handlers.
    if (
        $line =~ /<ServiceConfig(?:\s|>)/
        && $line =~ /MailFetcherService/
    ) {
        $mailfetcher_deleted++;
        next;
    }

    # Replace Prod hostname only in text gadget preferences.
    if (
        $line =~ /<GadgetUserPreference(?:\s|>)/
        && index($line, $prod_host) >= 0
    ) {
        $gadget_url_replaced +=
            ($line =~ s/\Q$prod_host\E/$qa_host/g);
    }

    # Replace the exact attachment path if it is persisted in the XML.
    if (
        $prod_attach ne $qa_attach
        && index($line, $prod_attach) >= 0
    ) {
        $attach_path_replaced +=
            ($line =~ s/\Q$prod_attach\E/$qa_attach/g);
    }

    print $line;
}

END {
    print STDERR "DELETED_CLUSTERNODE=$cluster_deleted\n";
    print STDERR "DELETED_MAILSERVER=$mailserver_deleted\n";
    print STDERR "DELETED_MAILFETCHER=$mailfetcher_deleted\n";
    print STDERR "GADGET_URL_REPLACEMENTS=$gadget_url_replaced\n";
    print STDERR "ATTACH_PATH_REPLACEMENTS=$attach_path_replaced\n";
    print STDERR "FILEATTACHMENT_IN=$fileattachment_count\n";
}
PERL
