  --cert itsm_opensearch_qa.pem \
  --key itsm_opensearch_qa.key \
  -H 'Content-Type: application/json' \
  -d '{
    "test": "CCR-Continuous",

    "version": 1
  }'

  'https://opensearch-itsm-qa-dr.srv.ms.com.cn:9200/ccr-itsm-qa/_search?size=10' \

        [.hits.hits[] | {
